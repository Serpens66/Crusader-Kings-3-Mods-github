        show_as_unavailable = {
            OR = {
                gold < AE_costs_based_on_diplomacy_skill
                prestige < AE_costs_based_on_diplomacy_skill

                    OR = {
                        has_character_modifier = additional_education_study_martial_modifier

                }
            }
		}
        scope:actor = {
            add_stress = 10 # we as actor also get a bit of stress, so we can not use it too often at too many people at once

            add_prestige = { subtract = AE_costs_based_on_diplomacy_skill }
		}
        name = additional_education.0007.name		#"Diplomacy"
		custom_tooltip = additional_education.0007.tooltip

			add_stress = 35

                days = 367

            add_opinion = { 
                modifier = payed_studies_opinion

            }
            if = {

additional_education.0003 = {
	type = character_event
	title = additional_education.1001.t
	desc = additional_education.1001.desc	
	theme = education
	
	left_portrait = {
		character = scope:actor
		animation = idle
	}

	right_portrait = {
		character = scope:recipient
		animation = personality_zealous
	}
    
    immediate = {
		# Martial        
				
		if = {
			limit = {
				scope:recipient = {
                    has_character_modifier = additional_education_study_martial_modifier
				}
			}
			scope:recipient = {
				add_character_modifier = {
                    modifier = additional_education_martial_modifier
                }
                remove_character_modifier = additional_education_study_martial_modifier
			}
		}

		# Stewardship

		else_if = {
			limit = {
				scope:recipient = {
                    has_character_modifier = additional_education_study_stewardship_modifier
				}
			}
			scope:recipient = {
				add_character_modifier = {
                    modifier = additional_education_stewardship_modifier
                }
                remove_character_modifier = additional_education_study_stewardship_modifier
			}
		}


		# Intrigue

		else_if = {
			limit = {
				scope:recipient = {
                    has_character_modifier = additional_education_study_intrigue_modifier
				}
			}
			scope:recipient = {
				add_character_modifier = {
                    modifier = additional_education_intrigue_modifier
                }
                remove_character_modifier = additional_education_study_intrigue_modifier
			}
		}

		# Diplomacy

		else_if = {
			limit = {
				scope:recipient = {
                    has_character_modifier = additional_education_study_diplomacy_modifier
				}
			}
			scope:recipient = {
				add_character_modifier = {
                    modifier = additional_education_diplomacy_modifier
                }
                remove_character_modifier = additional_education_study_diplomacy_modifier
			}
		}	


		# Learning

		else_if = {
			limit = {
				scope:recipient = {
                    has_character_modifier = additional_education_study_learning_modifier
				}
			}
			scope:recipient = {
				add_character_modifier = {
                    modifier = additional_education_learning_modifier
                }
                remove_character_modifier = additional_education_study_learning_modifier
			}
		}
        
        # prowess

		else_if = {
			limit = {
				scope:recipient = {
                    has_character_modifier = additional_education_study_prowess_modifier
				}
			}
			scope:recipient = {
				add_character_modifier = {
                    modifier = additional_education_prowess_modifier
                }
                remove_character_modifier = additional_education_study_prowess_modifier
			}
		}
	}
    
	option = {	
		name = additional_education.1001.name # just confirm what already happend
    }
