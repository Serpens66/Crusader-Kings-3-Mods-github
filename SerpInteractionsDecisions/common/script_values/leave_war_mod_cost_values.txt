

LW_prestige_costs_based_on_title_of_actor_value = {
	scope:actor = {
        value = 50
        multiply = highest_held_title_tier
        if = {
            limit = { 
                has_game_rule = leave_war_mod_costs_half
            }
            divide = 2
        }
        else_if = {
            limit = { 
                has_game_rule = leave_war_mod_costs_double
            }
            multiply = 2
        }
    }    
}
