kmc_toggle_dynasty_cannot_be_knight = {
    scope = character
    is_shown = {
        NOT = {
            has_character_flag = kmc_dynasty_cannot_be_knight
        }
    }
    effect = {
        if = {
            limit = {
                NOT = {
                    has_character_flag = kmc_dynasty_cannot_be_knight
                }
            }
            add_character_flag = kmc_dynasty_cannot_be_knight
        }
        else = {
            remove_character_flag  =  kmc_dynasty_cannot_be_knight
        }

    }
}

# the following uses variable, not global variable, these are also shared between all players, but attached to a scope, so basically the same like characterflag, while the latter can only be attached to characters.
 # it makes sense that this is already used here, because when your character changes, you want to reset the manual filter, since you will get totally different people. 
 # so no need to replace it with characterflag
#Manually allowed
kmc_toggle_manually_allowed = {
    scope = character
    is_shown = {
        NOT = {
            has_variable = kmc_manually_allowed
        }
    }
    effect = {
        if = {
            limit = {
                has_variable = kmc_manually_allowed
            }
            remove_variable = kmc_manually_allowed
        }
        else = {
            set_variable  =  kmc_manually_allowed
        }
    }
}
