# an alert for all childs you can define education focus, if not already done
serp_action_set_education_child = {
    combine_into_one = yes
	check_create_action = {
        every_child = { # even if they are not on my court or my ward, I can choose focus for them
            if = {
                limit = {
                    is_adult = no
                    age >= childhood_personality_age # only at this age, you know what they are best in
                    # age < childhood_education_start_age # until that age you can change it, although this rule is not always valid it seams...
                    has_any_focus = no # if no focus was set already
                }
                save_temporary_scope_as = the_child
                try_create_important_action = {
                    important_action_type = serp_action_set_education_child
                    actor = root
                    recipient = scope:the_child
                }
             }
        }
        every_courtier_or_guest = { # even if they are not my child, I can choose focus for them
            if = {
                limit = {
                    is_adult = no
                    age >= childhood_personality_age # only at this age, you know what they are best in
                    # age < childhood_education_start_age # until that age you can change it, although this rule is not always valid it seams...
                    has_any_focus = no # if no focus was set already
                }
                save_temporary_scope_as = the_child
                try_create_important_action = {
                    important_action_type = serp_action_set_education_child
                    actor = root
                    recipient = scope:the_child
                }
            }
        }
	}
	effect = {
		scope:recipient = {
            open_view_data = {
				view = character_focus # open focus window (for childs it is education)
                player = root
			}
        }
        scope:recipient = {
            open_view_data = {
				view = character # open character window
                player = root
			}
        }
	}
}

# demand conversion. for vassals we can not check the chances. But for courtiers only show who will accept
serp_action_demand_conversion_vassal = {
	combine_into_one = yes
    priority = 15
	check_create_action = {
        save_temporary_scope_as = actor
		every_vassal_or_below = {
			save_temporary_scope_as = recipient
            if = {
                limit = {
                    root = {
                        is_character_interaction_potentially_accepted = {
                            recipient = prev
                            interaction = demand_conversion_vassal_ruler_interaction
                        }
                    }
                    # demand_conversion_likelihood_calculation > 99 # is not the same like the shown percentage... but both, this and the percentage are not very accurate anyway it seems, so we will display everyone instead.
                }
                try_create_important_action = {
                    important_action_type = serp_action_demand_conversion_vassal
                    actor = root
                    recipient = this
                    secondary_recipient = this
                }
            }
		}
	}

	effect = {
		open_interaction_window = {
			interaction = demand_conversion_vassal_ruler_interaction
			actor = scope:actor
			recipient = scope:recipient
			secondary_recipient = scope:secondary_recipient
		}
	}
}
