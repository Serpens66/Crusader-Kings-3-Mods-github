# open an event to choose the war to leave
# (even better would be using the special_interaction = interfere_in_war_interaction GUI, but this is hardcoded to not allow to pick wars you are in.. no clue how to change it or make my own special_interaction )

leave_war_interaction_mod = {
	category = interaction_category_diplomacy
		
	desc = leave_war_interaction_mod_desc
	
	auto_accept = yes
    
    ai_will_do = {
		base = 0 #The AI should never do this!
	}
	

    # Actor-only availability also prevents scripted AI/Adventurer use.
    is_available = {
        is_alive = yes
        is_ai = no
        is_at_war = yes
        is_landless_adventurer = no
    }

    is_shown = {
        scope:recipient = {
            is_at_war = yes
            is_allied_in_war = scope:actor
            any_character_war = {
                lw_mod_can_leave_war_trigger = yes
            }
        }
        
    }
    
	on_accept = {
        scope:actor = {
			custom_tooltip = leave_war_interaction_mod_desc
		}
	}

    
    on_send = {
        lw_mod_clear_war_scopes_effect = yes # Discard any earlier picker references.
        scope:recipient = {
            every_character_war = {
                limit = {
                    lw_mod_can_leave_war_trigger = yes
                }
                if = { # save all the shared wars to the scope, so we can make an option for each of it within event. Allow up to 10.
                    limit = {
                        NOT = { exists = scope:leaving_war_1 }
                    }
                    save_scope_as = leaving_war_1
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        NOT = { exists = scope:leaving_war_2 }
                    }
                    save_scope_as = leaving_war_2
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        NOT = { exists = scope:leaving_war_3 }
                    }
                    save_scope_as = leaving_war_3
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        NOT = { exists = scope:leaving_war_4 }
                    }
                    save_scope_as = leaving_war_4
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        exists = scope:leaving_war_4
                        NOT = { exists = scope:leaving_war_5 }
                    }
                    save_scope_as = leaving_war_5
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        exists = scope:leaving_war_4
                        exists = scope:leaving_war_5
                        NOT = { exists = scope:leaving_war_6 }
                    }
                    save_scope_as = leaving_war_6
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        exists = scope:leaving_war_4
                        exists = scope:leaving_war_5
                        exists = scope:leaving_war_6
                        NOT = { exists = scope:leaving_war_7 }
                    }
                    save_scope_as = leaving_war_7
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        exists = scope:leaving_war_4
                        exists = scope:leaving_war_5
                        exists = scope:leaving_war_6
                        exists = scope:leaving_war_7
                        NOT = { exists = scope:leaving_war_8 }
                    }
                    save_scope_as = leaving_war_8
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        exists = scope:leaving_war_4
                        exists = scope:leaving_war_5
                        exists = scope:leaving_war_6
                        exists = scope:leaving_war_7
                        exists = scope:leaving_war_8
                        NOT = { exists = scope:leaving_war_9 }
                    }
                    save_scope_as = leaving_war_9
                }
                else_if = {
                    limit = {
                        exists = scope:leaving_war_1
                        exists = scope:leaving_war_2
                        exists = scope:leaving_war_3
                        exists = scope:leaving_war_4
                        exists = scope:leaving_war_5
                        exists = scope:leaving_war_6
                        exists = scope:leaving_war_7
                        exists = scope:leaving_war_8
                        exists = scope:leaving_war_9
                        NOT = { exists = scope:leaving_war_10 }
                    }
                    save_scope_as = leaving_war_10
                }
            }
        }
        
        scope:actor  = {
			trigger_event = {
				id = leave_war_mod.0001
			}
		}
		
	}
}
