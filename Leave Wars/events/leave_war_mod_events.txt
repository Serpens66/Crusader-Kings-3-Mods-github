namespace = leave_war_mod


leave_war_mod.0001 = {
	type = character_event
	title = leave_war_interaction_mod
	desc = leave_war_mod_event_desc
	theme = war
	
	left_portrait= {
		character = scope:actor
		animation = personality_callous
	}

	right_portrait = {
		character = scope:recipient
		animation = personality_bold
	}	
	   
    option = { # we have to hardcode options, we can not simply add an option per active war or so -.-. So I will hardcode ~ 10 options. I hope up to 10 same wars with the same person should be enough...
        name = leave_war_mod_optionname_1
        show_as_unavailable = { 
            lw_mod_can_select_war_trigger = { WAR = scope:leaving_war_1 }
            AND = {
                OR = {
                    has_game_rule = leave_war_mod_costs_default
                    has_game_rule = leave_war_mod_costs_half
                    has_game_rule = leave_war_mod_costs_double
                    has_game_rule = leave_war_mod_costs_only_prestigehonorgold
                }
                OR = {
                    prestige < LW_prestige_costs_based_on_title_of_actor_value
                    gold < LW_gold_costs_based_on_title_of_actor_value
                }
            }
        }
        trigger = {
            lw_mod_can_select_war_trigger = { WAR = scope:leaving_war_1 }
            OR = {
                has_game_rule = leave_war_mod_costs_nothing
                has_game_rule = leave_war_mod_costs_only_opinion
                AND = {
                    prestige >= LW_prestige_costs_based_on_title_of_actor_value
                    gold >= LW_gold_costs_based_on_title_of_actor_value
                }
            }
            
        }
        # Recheck after the player has left the picker open; no stale paid no-op.
        if = {
            limit = {
                lw_mod_can_select_war_trigger = { WAR = scope:leaving_war_1 }
                lw_mod_can_pay_war_exit_trigger = yes
            }
            scope:actor = {
                if = { 
                    limit = { 
                        OR = {
                            has_game_rule = leave_war_mod_costs_default
                            has_game_rule = leave_war_mod_costs_half
                            has_game_rule = leave_war_mod_costs_double
                            has_game_rule = leave_war_mod_costs_only_prestigehonorgold
                        }
                    }
                    add_prestige = { subtract = LW_prestige_costs_based_on_title_of_actor_value }
                    add_prestige_experience = { subtract = LW_prestige_costs_based_on_title_of_actor_value }
                    pay_short_term_gold = {
                        gold = LW_gold_costs_based_on_title_of_actor_value
                        target = scope:recipient
                    }
                }
                if = {
                    limit = {
                        is_allied_to = scope:recipient
                    }
                    break_alliance = scope:recipient # break the alliance, if there is one
                }
            }
            scope:leaving_war_1 = {
                save_scope_as = leaving_war
                lw_mod_war_departure_commitments_effect = yes
                custom_description = {
                    text = leave_war_mod_remove_participant # custom description for this effect, because oterhwise:  for whatever reason this triggers "REMOVE_PARTICIPANT_EFFECT_THIRD" instead of "REMOVE_PARTICIPANT_EFFECT_FIRST" in the tooltip of event, causing ALOT of error.log entries about missing character. WAR is known, but CHARACTER not. I think this is a bug somewhere in the devs code to trigger the wrong tooltip.
                    remove_participant = scope:actor
                }
                scope:recipient = {
                    if = {
                        limit = {
                            OR = {
                                has_game_rule = leave_war_mod_costs_default
                                has_game_rule = leave_war_mod_costs_half
                                has_game_rule = leave_war_mod_costs_double
                                has_game_rule = leave_war_mod_costs_only_opinion
                            }
                        }
                        add_opinion = {
                            target = scope:actor
                            modifier = left_my_war_opinion_modifier # recipient/leader gets very bad opinion about actor
                            opinion = LW_opinion_costs_leaders
                        }
                    }
                    if = {
                        limit = {
                            OR = {
                                has_game_rule = leave_war_mod_costs_default
                                has_game_rule = leave_war_mod_costs_half
                                has_game_rule = leave_war_mod_costs_double
                                has_game_rule = leave_war_mod_costs_only_prestigehonorgold
                            }
                        }
                        add_prestige = LW_prestige_costs_based_on_title_of_actor_value # the recipient is getting the amount of prestige
                    }
                }
                if = {
                    limit = {
                        is_defender = scope:recipient
                    }
                    every_war_defender = {
                        if = {
                            limit = {
                                NOR = {
                                    this = scope:actor
                                    this = scope:recipient
                                }
                                OR = {
                                    has_game_rule = leave_war_mod_costs_default
                                    has_game_rule = leave_war_mod_costs_half
                                    has_game_rule = leave_war_mod_costs_double
                                    has_game_rule = leave_war_mod_costs_only_opinion
                                }
                            }
                            add_opinion = {
                                target = scope:actor
                                modifier = left_participating_war_opinion_modifier # other defenders in the war get bad opinion about actor
                                opinion = LW_opinion_costs_participant
                            }
                        }
                        if = {
                            limit = {NOT = { this = scope:actor } }
                            send_interface_message = { # ally in war is leaving war message. normally the game should send it, but it seems buggy, so we will send it manually here
                                type = serp_msg_war_ally_removed
                                title = serp_serp_msg_war_ally_removed
                                desc = serp_serp_msg_war_ally_removed_desc
                                left_icon = scope:actor
                                right_icon = scope:recipient
                            }
                        }
                    }
                    every_war_attacker = {
                        send_interface_message = { # enemy is leaving war message. normally the game should send it, but it seems buggy, so we will send it manually here
                            type = serp_msg_war_enemy_removed
                            title = serp_serp_msg_war_enemy_removed
                            desc = serp_serp_msg_war_enemy_removed_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                        }
                    }
                }
                else_if = {
                    limit = {
                        is_attacker = scope:recipient
                    }
                    every_war_attacker = {
                        if = {
                            limit = {
                                NOR = {
                                    this = scope:actor
                                    this = scope:recipient
                                }
                                OR = {
                                    has_game_rule = leave_war_mod_costs_default
                                    has_game_rule = leave_war_mod_costs_half
                                    has_game_rule = leave_war_mod_costs_double
                                    has_game_rule = leave_war_mod_costs_only_opinion
                                }
                            }
                            add_opinion = {
                                target = scope:actor
                                modifier = left_participating_war_opinion_modifier # other attackers in the war get bad opinion about actor
                                opinion = LW_opinion_costs_participant
                            }
                        }
                        if = {
                            limit = {NOT = { this = scope:actor } }
                            send_interface_message = { # ally in war is leaving war message. normally the game should send it, but it seems buggy, so we will send it manually here
                                type = serp_msg_war_ally_removed
                                title = serp_serp_msg_war_ally_removed
                                desc = serp_serp_msg_war_ally_removed_desc
                                left_icon = scope:actor
                                right_icon = scope:recipient
                            }
                        }
                    }
                    every_war_defender = {
                        send_interface_message = { # enemy is leaving war message. normally the game should send it, but it seems buggy, so we will send it manually here
                            type = serp_msg_war_enemy_removed
                            title = serp_serp_msg_war_enemy_removed
                            desc = serp_serp_msg_war_enemy_removed_desc
                            left_icon = scope:actor
                            right_icon = scope:recipient
                        }
                    }
                }
            
            }
            scope:actor = { # after leaving_war was set
                send_interface_message = { # you are leaving war message. normally the game should send it, but it seems buggy, so we will send it manually here
                    type = serp_msg_war_player_removed
                    title = serp_serp_msg_war_player_removed
                    desc = serp_serp_msg_war_player_removed_desc
                    left_icon = scope:actor
                    right_icon = scope:recipient
                }
            }
        }
        clear_saved_scope = leaving_war
        clear_saved_scope = leaving_war_1
        clear_saved_scope = leaving_war_2
        clear_saved_scope = leaving_war_3
        clear_saved_scope = leaving_war_4
